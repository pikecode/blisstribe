# 商城模块第一阶段实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现平台自营商品的完整交易闭环（商品 → 购物车 → 下单 → 支付 → 发货 → 退款）

**Architecture:** 模块化单体，遵循现有 NestJS 模式。核心模块：shop.product → shop.cart → shop.order → shop.payment → shop.inventory → shop.refund。库存以 PostgreSQL 事务为事实来源，Redis 仅用缓存和限流。支付采用微信支付 V3，第一阶段资金先进平台账户，不实现自动分账。

**Tech Stack:** NestJS 10.x, Prisma 5.x, PostgreSQL, Redis (ioredis), 微信支付 SDK, Jest (单元测试), SuperTest (集成测试)

**Spec:** `docs/shop-module-design-optimized.md`

---

## 全局约束

- 金额统一使用"分"（Integer），禁止 Float
- 所有订单查询必须按 `userId` 过滤（用户隔离）
- 后台接口必须使用 `AdminJwtGuard`
- 库存以 PostgreSQL 事务为准
- 微信支付回调必须验签和幂等
- 金额、库存、商品状态全部服务端验证
- 商品详情只返回已上架且未删除商品
- 支付订单过期时间：15 分钟
- 退款金额不得超过已支付未退款金额
- 生产配置无默认密码和默认密钥

// apps/api/src/shop/category/category.service.ts
import { Injectable, ConflictException, NotFoundException } from '@nestjs/common'
import { CategoryRepository } from './category.repository'
import { CreateCategoryDto, UpdateCategoryDto } from '../dto/category.dto'

@Injectable()
export class CategoryService {
  constructor(private categoryRepository: CategoryRepository) {}

  async getAllCategories() {
    return this.categoryRepository.findAll()
  }

  async createCategory(dto: CreateCategoryDto) {
    // 检查 code 是否已存在
    const existing = await this.categoryRepository.findByCode(dto.code)
    if (existing) {
      throw new ConflictException('分类编码已存在')
    }

    return this.categoryRepository.create({
      code: dto.code,
      name: dto.name,
      sortOrder: dto.sortOrder || 0,
    })
  }

  async updateCategory(id: bigint, dto: UpdateCategoryDto) {
    const category = await this.categoryRepository.findById(id)
    if (!category) {
      throw new NotFoundException('分类不存在')
    }

    return this.categoryRepository.update(id, {
      name: dto.name,
      sortOrder: dto.sortOrder,
      status: dto.status,
    })
  }

  async deleteCategory(id: bigint) {
    const category = await this.categoryRepository.findById(id)
    if (!category) {
      throw new NotFoundException('分类不存在')
    }

    return this.categoryRepository.delete(id)
  }
}

@Injectable()
export class ProductRepository {

  async findById(id: bigint) {
    return this.prisma.shopProduct.findUnique({
      where: { id },
      include: { category: true },
    })
  }

  async findByIdForUpdate(id: bigint) {
    // 用于事务中的 for update
    return this.prisma.shopProduct.findUnique({
      where: { id },
    })
  }

  async findActiveById(id: bigint) {
    return this.prisma.shopProduct.findFirst({
      where: {
        id,
        status: 1, // 只返回已上架商品
        deletedAt: null,
      },
      include: { category: true },
    })
  }

  async findMany(filters: {
    categoryId?: bigint
    status?: number
    search?: string
    limit?: number
    offset?: number
  }) {
    const { categoryId, status, search, limit = 20, offset = 0 } = filters

    const where: any = { deletedAt: null }
    if (categoryId) where.categoryId = categoryId
    if (status !== undefined) where.status = status
    if (search) {
      where.OR = [
        { name: { contains: search, mode: 'insensitive' } },
        { description: { contains: search, mode: 'insensitive' } },
      ]
    }

    const [items, total] = await Promise.all([
      this.prisma.shopProduct.findMany({
        where,
        include: { category: true },
        orderBy: [{ sortOrder: 'asc' }, { createdAt: 'desc' }],
        take: limit,
        skip: offset,
      }),
      this.prisma.shopProduct.count({ where }),
    ])

    return { items, total }
  }

  async create(data: any) {
    return this.prisma.shopProduct.create({
      data,
      include: { category: true },
    })
  }

  async update(id: bigint, data: any) {
    return this.prisma.shopProduct.update({
      where: { id },
      data,
      include: { category: true },
    })
  }

  async publish(id: bigint) {
    return this.update(id, { status: 1 })
  }

  async unpublish(id: bigint) {
    return this.update(id, { status: 2 })
  }

  async softDelete(id: bigint) {
    return this.update(id, { deletedAt: new Date() })
  }

// apps/api/src/shop/product/product.service.ts
import {
  Injectable,
  NotFoundException,
  BadRequestException,
} from '@nestjs/common'
import { ProductRepository } from './product.repository'
import { CategoryService } from '../category/category.service'
import { CreateProductDto, UpdateProductDto } from '../dto/product.dto'

@Injectable()
export class ProductService {
  constructor(
    private productRepository: ProductRepository,
    private categoryService: CategoryService
  ) {}

  async getProductList(filters: {
    categoryId?: number
    status?: number
    search?: string
    page?: number
    limit?: number
  }) {
    const { categoryId, status, search, page = 1, limit = 20 } = filters
    const offset = (page - 1) * limit

    return this.productRepository.findMany({
      categoryId: categoryId ? BigInt(categoryId) : undefined,
      status,
      search,
      limit,
      offset,
    })
  }

  async getPublicProductList(filters: {
    categoryId?: number
    search?: string
    page?: number
    limit?: number
  }) {
    // 仅返回已上架商品
    return this.getProductList({
      ...filters,
      status: 1,
    })
  }

  async getProductDetail(id: bigint) {
    const product = await this.productRepository.findActiveById(id)
    if (!product) {
      throw new NotFoundException('商品不存在或已下架')
    }
    return product
  }

  async createProduct(dto: CreateProductDto) {
    // 验证分类存在
    const category = await this.categoryService.getAllCategories()
    const categoryExists = category.some(c => c.id === BigInt(dto.categoryId))
    if (!categoryExists) {
      throw new NotFoundException('分类不存在')
    }

    // 验证价格和库存
    if (dto.priceFen <= 0) {
      throw new BadRequestException('价格必须大于0')
    }
    if (dto.totalStock < 0) {
      throw new BadRequestException('库存不能为负')
    }

    return this.productRepository.create({
      categoryId: BigInt(dto.categoryId),
      name: dto.name,
      description: dto.description || null,
      images: dto.images,
      priceFen: dto.priceFen,
      totalStock: dto.totalStock,
      reservedStock: 0,
      soldStock: 0,
      status: 0, // 默认草稿
    })
  }

  async updateProduct(id: bigint, dto: UpdateProductDto) {
    const product = await this.productRepository.findById(id)
    if (!product) {
      throw new NotFoundException('商品不存在')
    }

    // 验证修改的字段
    if (dto.priceFen !== undefined && dto.priceFen <= 0) {
      throw new BadRequestException('价格必须大于0')
    }
    if (dto.totalStock !== undefined && dto.totalStock < 0) {
      throw new BadRequestException('库存不能为负')
    }

    const updateData: any = {}
    if (dto.name !== undefined) updateData.name = dto.name
    if (dto.description !== undefined) updateData.description = dto.description
    if (dto.images !== undefined) updateData.images = dto.images
    if (dto.priceFen !== undefined) updateData.priceFen = dto.priceFen
    if (dto.totalStock !== undefined) updateData.totalStock = dto.totalStock
    if (dto.sortOrder !== undefined) updateData.sortOrder = dto.sortOrder
    if (dto.status !== undefined) updateData.status = dto.status

    return this.productRepository.update(id, updateData)
  }

  async publishProduct(id: bigint) {
    const product = await this.productRepository.findById(id)
    if (!product) {
      throw new NotFoundException('商品不存在')
    }
    return this.productRepository.publish(id)
  }

  async unpublishProduct(id: bigint) {
    const product = await this.productRepository.findById(id)
    if (!product) {
      throw new NotFoundException('商品不存在')
    }
    return this.productRepository.unpublish(id)
  }

  async deleteProduct(id: bigint) {
    const product = await this.productRepository.findById(id)
    if (!product) {
      throw new NotFoundException('商品不存在')
    }
    return this.productRepository.softDelete(id)
  }
}

// apps/api/src/shop/cart/cart.service.ts
import { Injectable, BadRequestException } from '@nestjs/common'
import { PrismaService } from '../../common/prisma.service'
import { ProductService } from '../product/product.service'
import { AddCartItemDto, UpdateCartItemDto } from '../dto/cart.dto'
import { AmountUtil } from '../common/utils/amount.util'

@Injectable()
export class CartService {
  constructor(
    private prisma: PrismaService,
    private productService: ProductService
  ) {}

  async getCart(userId: bigint) {
    let cart = await this.prisma.shopCart.findUnique({
      where: { userId },
      include: {
        items: {
          include: { product: true },
        },
      },
    })

    // 购物车不存在，创建空购物车
    if (!cart) {
      cart = await this.prisma.shopCart.create({
        data: { userId },
        include: { items: { include: { product: true } } },
      })
    }

    // 计算合计
    const totalAmount = cart.items.reduce(
      (sum, item) => sum + item.product.priceFen * item.quantity,
      0
    )

    return {
      items: cart.items.map(item => ({
        id: item.id,
        productId: item.productId,
        quantity: item.quantity,
        product: {
          id: item.product.id,
          name: item.product.name,
          images: item.product.images,
          priceFen: item.product.priceFen,
          totalStock: item.product.totalStock,
          available:
            item.product.totalStock -
            item.product.reservedStock -
            item.product.soldStock,
        },
      })),
      totalAmount,
      totalQuantity: cart.items.reduce((sum, item) => sum + item.quantity, 0),
    }
  }

  async addItem(userId: bigint, dto: AddCartItemDto) {
    // 验证商品存在且上架
    const product = await this.productService.getProductDetail(
      BigInt(dto.productId)
    )

    // 验证库存
    const available =
      product.totalStock - product.reservedStock - product.soldStock
    if (available < dto.quantity) {
      throw new BadRequestException(
        `库存不足，仅剩${available}件`
      )
    }

    // 获取或创建购物车
    let cart = await this.prisma.shopCart.findUnique({
      where: { userId },
    })
    if (!cart) {
      cart = await this.prisma.shopCart.create({
        data: { userId },
      })
    }

    // 更新或创建购物车项
    const existingItem = await this.prisma.shopCartItem.findUnique({
      where: {
        cartId_productId: {
          cartId: cart.id,
          productId: BigInt(dto.productId),
        },
      },
    })

    if (existingItem) {
      // 检查新数量是否超库存
      const newQuantity = existingItem.quantity + dto.quantity
      if (newQuantity > available) {
        throw new BadRequestException(
          `库存不足，最多可添加${available - existingItem.quantity}件`
        )
      }

      return this.prisma.shopCartItem.update({
        where: { id: existingItem.id },
        data: { quantity: newQuantity },
        include: { product: true },
      })
    }

    return this.prisma.shopCartItem.create({
      data: {
        cartId: cart.id,
        productId: BigInt(dto.productId),
        quantity: dto.quantity,
      },
      include: { product: true },
    })
  }

  async updateItem(userId: bigint, itemId: bigint, dto: UpdateCartItemDto) {
    const item = await this.prisma.shopCartItem.findUnique({
      where: { id: itemId },
      include: { cart: true, product: true },
    })

    if (!item || item.cart.userId !== userId) {
      throw new BadRequestException('购物车项不存在')
    }

    if (dto.quantity <= 0) {
      throw new BadRequestException('数量必须大于0')
    }

    // 检查库存
    const available =
      item.product.totalStock -
      item.product.reservedStock -
      item.product.soldStock
    if (dto.quantity > available) {
      throw new BadRequestException(
        `库存不足，仅剩${available}件`
      )
    }

    return this.prisma.shopCartItem.update({
      where: { id: itemId },
      data: { quantity: dto.quantity },
      include: { product: true },
    })
  }

  async removeItem(userId: bigint, itemId: bigint) {
    const item = await this.prisma.shopCartItem.findUnique({
      where: { id: itemId },
      include: { cart: true },
    })

    if (!item || item.cart.userId !== userId) {
      throw new BadRequestException('购物车项不存在')
    }

// apps/api/src/shop/order/order.service.ts
import {
  Injectable,
  BadRequestException,
  NotFoundException,
} from '@nestjs/common'
import { PrismaService } from '../../common/prisma.service'
import { OrderRepository } from './order.repository'
import { ProductService } from '../product/product.service'
import { CartService } from '../cart/cart.service'
import { OrderNoGenerator } from './order-no.generator'
import { CreateOrderDto } from '../dto/order.dto'

@Injectable()
export class OrderService {
  constructor(
    private prisma: PrismaService,
    private orderRepository: OrderRepository,
    private productService: ProductService,
    private cartService: CartService
  ) {}

  async createOrder(userId: bigint, dto: CreateOrderDto) {
    // 验证收货信息
    if (!dto.receiverName || !dto.receiverPhone || !dto.shippingAddress) {
      throw new BadRequestException('收货信息不完整')
    }

    if (dto.items.length === 0) {
      throw new BadRequestException('订单至少包含一个商品')
    }

    // 使用事务原子创建订单
    const order = await this.prisma.$transaction(async (tx) => {
      let totalAmountFen = 0
      const orderItems = []

      // 逐个校验和预留商品
      for (const cartItem of dto.items) {
        const product = await tx.shopProduct.findUnique({
          where: { id: BigInt(cartItem.productId) },
        })

        if (!product) {
          throw new NotFoundException(`商品 ${cartItem.productId} 不存在`)
        }

        // 校验商品上架状态
        if (product.status !== 1 || product.deletedAt) {
          throw new BadRequestException(
            `商品"${product.name}"已下架`
          )
        }

        // 校验库存
        const available =
          product.totalStock - product.reservedStock - product.soldStock
        if (available < cartItem.quantity) {
          throw new BadRequestException(
            `商品"${product.name}"库存不足，仅剩${available}件`
          )
        }

        // 预留库存
        await tx.shopProduct.update({
          where: { id: product.id },
          data: {
            reservedStock: { increment: cartItem.quantity },
          },
        })

        const subtotalFen = product.priceFen * cartItem.quantity
        totalAmountFen += subtotalFen

        orderItems.push({
          productName: product.name,
          productImage: product.images[0] || null,
          unitPriceFen: product.priceFen,
          quantity: cartItem.quantity,
          subtotalFen,
          productId: product.id,
        })
      }

      // 创建订单
      const orderNo = OrderNoGenerator.generate()
      const expiresAt = new Date(Date.now() + 15 * 60 * 1000) // 15分钟后过期

      const newOrder = await tx.shopOrder.create({
        data: {
          orderNo,
          userId,
          status: 'pending_payment',
          paymentStatus: 'unpaid',
          fulfillmentStatus: 'pending',
          totalAmountFen,
          paymentAmountFen: totalAmountFen,
          discountAmountFen: 0,
          receiverName: dto.receiverName,
          receiverPhone: dto.receiverPhone,
          shippingAddress: dto.shippingAddress,
          remark: dto.remark || null,
          expiresAt,
          items: {
            create: orderItems,
          },
        },
        include: { items: true },
      })

      return newOrder
    })

    // 清空购物车
    await this.cartService.clearCart(userId)

    return order
  }

  async getOrderDetail(orderId: bigint, userId: bigint) {
    const order = await this.orderRepository.findById(orderId)
    if (!order || order.userId !== userId) {
      throw new NotFoundException('订单不存在')
    }
    return order
  }

  async getUserOrders(userId: bigint, filters: any = {}) {
    return this.orderRepository.findUserOrders(userId, filters)
  }

  async cancelOrder(orderId: bigint, userId: bigint) {
    const order = await this.orderRepository.findById(orderId)
    if (!order || order.userId !== userId) {
      throw new NotFoundException('订单不存在')
    }

    if (order.status !== 'pending_payment') {
      throw new BadRequestException('只能取消未支付订单')
    }

    // 事务中释放库存
    await this.prisma.$transaction(async (tx) => {
      for (const item of order.items) {
        await tx.shopProduct.update({
          where: { id: item.productId },
          data: {
            reservedStock: { decrement: item.quantity },
          },
        })
      }

      await tx.shopOrder.update({
        where: { id: orderId },
        data: {
          status: 'cancelled',
          cancelledAt: new Date(),
          cancelReason: 'User cancelled',
        },
      })
    })

    return { message: '订单已取消' }
  }
}

// apps/api/src/shop/payment/payment.service.ts
import { Injectable, BadRequestException } from '@nestjs/common'
import { PrismaService } from '../../common/prisma.service'
import { WechatPayService } from './wechat-pay.service'
import { OrderRepository } from '../order/order.repository'

@Injectable()
export class PaymentService {
  constructor(
    private prisma: PrismaService,
    private wechatPay: WechatPayService,
    private orderRepository: OrderRepository
  ) {}

  async createPayment(orderId: bigint, clientIp: string) {
    const order = await this.orderRepository.findById(orderId)
    if (!order) {
      throw new BadRequestException('订单不存在')
    }

    if (order.paymentStatus !== 'unpaid') {
      throw new BadRequestException('订单不支持重复支付')
    }

    // 调用微信统一下单
    const { prepayId } = await this.wechatPay.createPrepay({
      outTradeNo: order.orderNo,
      amount: order.paymentAmountFen,
      description: `BlissTribe订单${order.orderNo}`,
      notifyUrl: `${process.env.API_BASE_URL}/shop/webhooks/wechat-pay`,
      clientIp,
    })

    // 保存支付记录
    const payment = await this.prisma.shopPayment.create({
      data: {
        orderId,
        outTradeNo: order.orderNo,
        prepayId,
        amountFen: order.paymentAmountFen,
        status: 'pending',
      },
    })

    return { prepayId, outTradeNo: order.orderNo }
  }

  // 处理微信回调
  async handleWechatNotify(data: any) {
    const { out_trade_no, transaction_id, amount } = data

    // 查找订单
    const order = await this.orderRepository.findByOrderNo(out_trade_no)
    if (!order) {
      throw new BadRequestException('订单不存在')
    }

    // 金额校验
    if (amount.total !== order.paymentAmountFen) {
      throw new BadRequestException('金额不匹配')
    }

    // 事务中更新支付状态
    await this.prisma.$transaction(async (tx) => {
      // 检查支付记录是否已处理（幂等）
      const payment = await tx.shopPayment.findFirst({
        where: { orderId: order.id, wechatTransactionId: transaction_id },
      })

      if (payment) {
        // 重复回调，直接返回
        return
      }

      // 更新支付记录
      await tx.shopPayment.update({
        where: { orderId: order.id },
        data: {
          wechatTransactionId: transaction_id,
          status: 'success',
          paidAt: new Date(),
        },
      })

      // 更新订单状态
      await tx.shopOrder.update({
        where: { id: order.id },
        data: {
          paymentStatus: 'paid',
          status: 'paid',
          paidAt: new Date(),
        },
      })

      // 库存扣减：reservedStock 转 soldStock
      for (const item of order.items) {
        await tx.shopProduct.update({
          where: { id: item.productId },
          data: {
            reservedStock: { decrement: item.quantity },
            soldStock: { increment: item.quantity },
          },
        })
      }
    })

    return { message: 'success' }
  }

  async processRefund(orderId: bigint, refundNo: string, amount: number) {
    const order = await this.orderRepository.findById(orderId)
    if (!order) {
      throw new BadRequestException('订单不存在')
    }

    const payment = order.payments[0]
    if (!payment?.wechatTransactionId) {
      throw new BadRequestException('订单未支付')
    }

    // 调用微信退款
    const { refundId } = await this.wechatPay.refund({
      transactionId: payment.wechatTransactionId,
      outRefundNo: refundNo,
      amount,
      reason: 'User requested',
    })

    return { refundId }
  }
}
